<script>
    document.addEventListener('DOMContentLoaded', function() {
        const countryCodeMap = { 'KOREA': 'kr', 'KR': 'kr', 'USA': 'us', 'US': 'us', 'JAPAN': 'jp', 'CHINA': 'cn', 'VIETNAM': 'vn' };
        const rawCountry = "{{ installbase.country|default:'' }}".toUpperCase().trim();
        const code = countryCodeMap[rawCountry];
        if (code) {
            const flagBase = `https://flagcdn.com/w80/${code}.png`;
            if (document.getElementById('detail-flag-display')) document.getElementById('detail-flag-display').innerHTML = `<img src="${flagBase}" class="w-full h-full object-cover">`;
            if (document.getElementById('spec-flag-icon')) document.getElementById('spec-flag-icon').innerHTML = `<img src="${flagBase}" class="w-full h-full object-cover">`;
        }

        window.toggleExpansionForm = function() { document.getElementById('expansionForm').classList.toggle('hidden'); };
        window.submitExpansion = function() {
            const date = document.getElementById('expDate').value;
            const description = document.getElementById('expDesc').value;
            if (!date || !description) return;
            fetch("{% url 'add_expansion_ajax' installbase.pk %}", {
                method: "POST",
                headers: { "X-CSRFToken": "{{ csrf_token }}", "Content-Type": "application/x-www-form-urlencoded" },
                body: `date=${encodeURIComponent(date)}&description=${encodeURIComponent(description)}`
            }).then(res => res.json()).then(data => { if(data.status === 'success') location.reload(); });
        };
        window.enableEdit = function(id) {
            const row = document.getElementById(`exp-row-${id}`);
            const dateCell = row.querySelector('.exp-date-cell');
            const descCell = row.querySelector('.exp-desc-cell');
            const actionCell = row.querySelector('td:last-child');
            const oldDate = dateCell.innerText.trim();
            const oldDesc = descCell.innerText.trim();
            dateCell.innerHTML = `<input type="date" id="edit-date-${id}" value="${oldDate}" class="w-full p-2 text-xs border rounded-lg">`;
            descCell.innerHTML = `<textarea id="edit-desc-${id}" rows="3" class="w-full p-2 text-xs border rounded-lg resize-none">${oldDesc}</textarea>`;
            actionCell.innerHTML = `<button onclick="saveEdit('${id}')" class="text-green-600 font-bold text-xs uppercase">Save</button>`;
        };
        window.saveEdit = function(id) {
            const newDate = document.getElementById(`edit-date-${id}`).value;
            const newDesc = document.getElementById(`edit-desc-${id}`).value;
            fetch(`/expansion/edit/${id}/`, {
                method: "POST",
                headers: { "X-CSRFToken": "{{ csrf_token }}", "Content-Type": "application/x-www-form-urlencoded" },
                body: `date=${encodeURIComponent(newDate)}&description=${encodeURIComponent(newDesc)}`
            }).then(res => res.json()).then(data => { if(data.status === 'success') location.reload(); });
        };
    });
</script>
