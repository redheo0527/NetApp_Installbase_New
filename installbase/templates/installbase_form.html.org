<script>
// 스위치 활성화/비활성화 로직
function toggleSwitchFields(isSwitchless) {
    const container = document.getElementById('switch-fields-container');
    const inputs = container.querySelectorAll('select, input');
    
    if (isSwitchless) {
        container.style.opacity = '0.4';
        container.style.pointerEvents = 'none';
        inputs.forEach(el => { el.disabled = true; el.value = ""; });
    } else {
        container.style.opacity = '1';
        container.style.pointerEvents = 'auto';
        inputs.forEach(el => el.disabled = false);
    }
}

function updateFileName(input) {
    const fileName = input.files[0] ? input.files[0].name : "선택된 파일 없음";
    const span = input.closest('div').querySelector('.file-name');
    span.textContent = fileName;
    span.classList.replace('text-gray-400', 'text-gray-800');
}

document.addEventListener('DOMContentLoaded', function() {
    // 실시간 이미지 업데이트
    async function updatePreview(selectElement, previewId, type) {
        const id = selectElement.value;
        const previewContainer = document.getElementById(previewId);
        if (!id) {
            previewContainer.innerHTML = `<i class="fa-solid fa-${type === 'customer' ? 'building' : 'box'} text-gray-100 text-5xl py-12"></i>`;
            return;
        }
        try {
            const response = await fetch(`/api/get-image-url/?type=${type}&id=${id}`);
            const data = await response.json();
            if (data.url) previewContainer.innerHTML = `<img src="${data.url}" class="w-full h-auto block animate-pop-in">`;
        } catch (e) { console.error(e); }
    }

    const customerSelect = document.querySelector('select[name="customer"]');
    const productSelect = document.querySelector('select[name="product"]');
    customerSelect.addEventListener('change', () => updatePreview(customerSelect, 'customer-logo-preview', 'customer'));
    productSelect.addEventListener('change', () => updatePreview(productSelect, 'product-image-preview', 'product'));

    window.openModal = (id) => { document.getElementById(id).classList.replace('hidden', 'flex'); };
    window.closeModal = (id) => { document.getElementById(id).classList.replace('flex', 'hidden'); };

    // AJAX 제출
    async function handleAjaxSubmit(formId, url, selectName, modalId, previewId, type) {
        const form = document.getElementById(formId);
        form.onsubmit = async (e) => {
            e.preventDefault();
            const formData = new FormData(form);
            const res = await fetch(url, { method: 'POST', body: formData, headers: { 'X-CSRFToken': '{{ csrf_token }}' } });
            const data = await res.json();
            if (res.ok) {
                const select = document.querySelector(`select[name="${selectName}"]`);
                select.add(new Option(data.name, data.id, true, true));
                closeModal(modalId);
                updatePreview(select, previewId, type);
                form.reset();
            }
        };
    }
    handleAjaxSubmit('customer-ajax-form', '/customer/add/ajax/', 'customer', 'customer-modal', 'customer-logo-preview', 'customer');
    handleAjaxSubmit('product-ajax-form', '/product/add/ajax/', 'product', 'product-modal', 'product-image-preview', 'product');

    // 엔지니어 풀
    const poolButtons = document.querySelectorAll('.engineer-source-btn');
    const badgeContainer = document.getElementById('engineer-badge-container');
    const hiddenSelect = document.querySelector('select[name="assigned_engineers"]');
    const placeholder = document.getElementById('placeholder-text');

    poolButtons.forEach(btn => {
        btn.addEventListener('click', function() {
            const userId = this.getAttribute('data-id');
            const userName = this.getAttribute('data-name');
            this.style.display = 'none';
            for (let opt of hiddenSelect.options) { if (opt.value === userId) opt.selected = true; }
            const badge = document.createElement('div');
            badge.className = "flex items-center gap-2 px-3 py-1.5 bg-[#0067B3] text-white rounded-full animate-pop-in mb-1 shadow-sm text-[11px]";
            badge.innerHTML = `<span>${userName}</span><button type="button" class="ml-1">×</button>`;
            badge.querySelector('button').onclick = () => {
                for (let opt of hiddenSelect.options) { if (opt.value === userId) opt.selected = false; }
                badge.remove();
                this.style.display = 'flex';
                if (badgeContainer.querySelectorAll('div').length === 0) placeholder.classList.remove('hidden');
            };
            badgeContainer.appendChild(badge);
            placeholder.classList.add('hidden');
        });
    });
});
</script>
