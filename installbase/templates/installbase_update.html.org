/**
 * [핵심] 스위치 타입 변경 시 Hidden Input 값과 UI 상태 동시 업데이트
 */
function updateSwitchType(isSwitchless) {
    const container = document.getElementById('switch-fields-container');
    const hiddenField = document.getElementById('id_is_switch_connected');
    const selects = container.querySelectorAll('select');
    
    if (isSwitchless) {
        // Switchless 선택 시
        container.style.opacity = '0.3';
        container.style.pointerEvents = 'none';
        hiddenField.value = "False"; // 서버 전송 값
        selects.forEach(el => { el.value = ""; }); // 선택 초기화
    } else {
        // Switched 선택 시
        container.style.opacity = '1';
        container.style.pointerEvents = 'auto';
        hiddenField.value = "True"; // 서버 전송 값
    }
}

// 담당 엔지니어 배지 삭제
function removeEngineerBadge(btn, userId) {
    const hiddenSelect = document.querySelector('select[name="assigned_engineers"]');
    const badgeContainer = document.getElementById('engineer-badge-container');
    const placeholder = document.getElementById('placeholder-text');

    for (let opt of hiddenSelect.options) {
        if (opt.value === userId) opt.selected = false;
    }

    btn.closest('.badge-item').remove();
    const sourceBtn = document.querySelector(`.engineer-source-btn[data-id="${userId}"]`);
    if(sourceBtn) sourceBtn.style.display = 'flex';

    if (badgeContainer.querySelectorAll('.badge-item').length === 0) {
        placeholder.classList.remove('hidden');
    }
}

document.addEventListener('DOMContentLoaded', function() {
    // 1. 국기 표시 초기화
    const countrySelect = document.querySelector('select[name="country"]');
    const flagDisplay = document.getElementById('country-flag-display');
    const countryCodeMap = {
        'KOREA': 'kr', 'SOUTH KOREA': 'kr', 'KR': 'kr', '한국': 'kr', '대한민국': 'kr',
        'USA': 'us', 'UNITED STATES': 'us', 'JAPAN': 'jp', 'CHINA': 'cn', 'VIETNAM': 'vn',
        'SINGAPORE': 'sg', 'GERMANY': 'de', 'INDIA': 'in', 'SLOVAKIA': 'sk'
    };

    function updateCountryFlag() {
        if (!countrySelect) return;
        const selectedText = countrySelect.options[countrySelect.selectedIndex]?.text.toUpperCase().trim();
        const countryCode = countryCodeMap[selectedText];
        flagDisplay.innerHTML = countryCode ? `<img src="https://flagcdn.com/w80/${countryCode}.png" class="w-full h-full object-cover">` : `<i class="fa-solid fa-earth-asia text-gray-200"></i>`;
    }
    if (countrySelect) { countrySelect.addEventListener('change', updateCountryFlag); updateCountryFlag(); }

    // 2. 엔지니어 추가 로직
    const poolButtons = document.querySelectorAll('.engineer-source-btn');
    const badgeContainer = document.getElementById('engineer-badge-container');
    const hiddenSelect = document.querySelector('select[name="assigned_engineers"]');
    const placeholder = document.getElementById('placeholder-text');

    poolButtons.forEach(btn => {
        btn.addEventListener('click', function() {
            const userId = this.getAttribute('data-id');
            const userName = this.getAttribute('data-name');
            this.style.display = 'none';
            for (let opt of hiddenSelect.options) { if (opt.value === userId) opt.selected = true; }

            const badge = document.createElement('div');
            badge.className = "flex items-center gap-2 px-3 py-1.5 bg-[#0067B3] text-white rounded-full text-[11px] badge-item animate-pop-in";
            badge.innerHTML = `<span>${userName}</span><button type="button" class="ml-1" onclick="removeEngineerBadge(this, '${userId}')">×</button>`;

            badgeContainer.appendChild(badge);
            placeholder.classList.add('hidden');
        });
    });

    // 3. 페이지 로드 시 스위치 초기 상태 설정
    const initialIsSwitchless = "{{ installbase.is_switch_connected }}" === "False";
    updateSwitchType(initialIsSwitchless);
});
</script>
